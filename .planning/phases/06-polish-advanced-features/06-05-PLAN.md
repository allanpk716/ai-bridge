```typescript
import { ComponentBoundary } from 'react-error-boundary';
import { AlertTriangle, RefreshCw } from 'lucide-react';
import { Button } from '@/components/ui/button';

function AppErrorFallback({
  error,
  resetErrorBoundary
}: {
  error: Error;
  resetErrorBoundary: () => void;
}) {
  return (
    <div className="flex items-center justify-center min-h-screen bg-background">
      <div className="max-w-md w-full mx-4 text-center">
        <AlertTriangle className="mx-auto h-16 w-16 text-destructive mb-4" />
        <h1 className="text-2xl font-bold mb-2">应用遇到了问题</h1>
        <p className="text-muted-foreground mb-6">
          {import.meta.env.DEV
            ? error.message
            : '抱歉,应用遇到了意外错误。请尝试刷新页面或重新加载。'}
        </p>
        <div className="flex gap-3 justify-center">
          <Button onClick={resetErrorBoundary} variant="default">
            <RefreshCw className="mr-2 h-4 w-4" />
            重试
          </Button>
          <Button
            onClick={() => window.location.reload()}
            variant="outline"
          >
            刷新页面
          </Button>
        </div>
        {import.meta.env.DEV && (
          <details className="mt-6 text-left">
            <summary className="cursor-pointer text-sm text-muted-foreground">
              查看错误详情
            </summary>
            <pre className="mt-2 p-4 bg-muted rounded text-xs overflow-auto">
              {error.stack}
            </pre>
          </details>
        )}
      </div>
    </div>
  );
}

export function AppErrorBoundary({ children }: { children: React.ReactNode }) {
  return (
    <ComponentBoundary
      FallbackComponent={AppErrorFallback}
      onError={(error) => {
        console.error('应用错误:', error);
        // 可选:发送到错误监控服务(如Sentry)
      }}
      onReset={() => {
        // 清除缓存、重置状态
        window.location.reload();
      }}
    >
      {children}
    </ComponentBoundary>
  );
}
```

**Verify:** ErrorBoundary在开发环境显示详细错误,生产环境显示友好提示

```typescript
import { QueryCache, MutationCache } from '@tanstack/react-query';
import { toast } from 'sonner';

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      retry: (failureCount, error) => {
        // 网络错误重试3次,其他错误不重试
        if (error instanceof NetworkError) {
          return failureCount < 3;
        }
        return false;
      },
      staleTime: 5 * 60 * 1000, // 5分钟
    },
    mutations: {
      retry: false,
    }
  },
  queryCache: new QueryCache({
    onError: (error) => {
      console.error('Query error:', error);
      toast.error(
        error.message || '加载数据失败,请检查网络连接'
      );
    }
  }),
  mutationCache: new MutationCache({
    onError: (error) => {
      console.error('Mutation error:', error);
      toast.error(
        error.message || '操作失败,请重试'
      );
    },
    onSuccess: () => {
      toast.success('操作成功');
    }
  })
});
```

### Task 4: 创建API错误处理工具

**File:** `src/lib/api/errorHandler.ts`

```typescript
export class APIError extends Error {
  constructor(
    message: string,
    public statusCode: number,
    public details?: unknown
  ) {
    super(message);
    this.name = 'APIError';
  }
}

export function handleAPIError(error: unknown): APIError {
  if (error instanceof APIError) {
    return error;
  }

  if (error instanceof Error) {
    // 网络错误
    if (error.message.includes('fetch')) {
      return new APIError(
        '网络连接失败,请检查您的网络',
        0
      );
    }

    // 超时错误
    if (error.message.includes('timeout')) {
      return new APIError(
        '请求超时,请稍后重试',
        408
      );
    }
  }

  return new APIError(
    '发生了未知错误,请稍后重试',
    500
  );
}

// 根据状态码生成用户友好消息
export function getStatusMessage(status: number): string {
  const messages: Record<number, string> = {
    400: '请求参数错误',
    401: '未授权,请重新登录',
    403: '没有权限访问',
    404: '请求的资源不存在',
    408: '请求超时',
    429: '请求过于频繁,请稍后重试',
    500: '服务器错误,请稍后重试',
    502: '网关错误',
    503: '服务暂时不可用',
  };

  return messages[status] || '发生了未知错误';
}
```

### Task 5: 创建重试Hook

**File:** `src/hooks/useRetry.ts`

```typescript
import { useState } from 'react';
import { toast } from 'sonner';

export function useRetry(
  fn: () => Promise<void>,
  options?: {
    maxAttempts?: number;
    delay?: number;
  }
) {
  const [attempt, setAttempt] = useState(0);
  const [isLoading, setIsLoading] = useState(false);
  const maxAttempts = options?.maxAttempts ?? 3;
  const delay = options?.delay ?? 1000;

  const retry = async () => {
    setIsLoading(true);
    try {
      await fn();
      setAttempt(0);
    } catch (error) {
      const newAttempt = attempt + 1;
      setAttempt(newAttempt);

      if (newAttempt < maxAttempts) {
        toast.error(
          `操作失败,${delay / 1000}秒后自动重试 (${newAttempt}/${maxAttempts})`
        );
        setTimeout(() => retry(), delay);
      } else {
        toast.error('操作失败,请手动重试');
      }
    } finally {
      setIsLoading(false);
    }
  };

  return { retry, attempt, isLoading };
}
```

```typescript
const mutation = useMutation({
  mutationFn: sendMessage,
  onMutate: async (newMessage) => {
    await queryClient.cancelQueries({ queryKey: ['messages'] });
    const previousMessages = queryClient.getQueryData(['messages']);
    queryClient.setQueryData(['messages'], (old: Message[]) => [
      ...old,
      { ...newMessage, status: 'sending' }
    ]);
    return { previousMessages };
  },
  onError: (err, newMessage, context) => {
    // 回滚到之前的状态
    queryClient.setQueryData(['messages'], context?.previousMessages);
    toast.error('消息发送失败');
  },
  onSettled: () => {
    queryClient.invalidateQueries({ queryKey: ['messages'] });
  }
});
```

```typescript
import { useOnlineStatus } from '@/hooks/useOnlineStatus';
import { WifiOff } from 'lucide-react';
import { toast } from 'sonner';

export function NetworkStatusError() {
  const isOnline = useOnlineStatus();

  useEffect(() => {
    if (!isOnline) {
      toast.error('网络连接已断开', {
        duration: Infinity,
        id: 'offline',
        action: {
          label: '重试',
          onClick: () => window.location.reload()
        }
      });
    } else {
      toast.dismiss('offline');
    }
  }, [isOnline]);

  if (isOnline) return null;

  return (
    <div className="fixed bottom-4 right-4 bg-destructive text-destructive-foreground px-4 py-3 rounded shadow-lg flex items-center gap-2">
      <WifiOff className="h-4 w-4" />
      <span>网络连接已断开</span>
    </div>
  );
}
```
